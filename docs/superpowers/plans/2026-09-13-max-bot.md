# Бот в MAX (бронь FBO + алерты ФБС) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a MAX-messenger bot that books FBO supply slots on Ozon (with a Wildberries
coefficient-alert fallback, since WB has no public booking endpoint) and monitors/alerts on
FBS shipment deadlines for both marketplaces, across multiple seller cabinets.

**Architecture:** New `app/bot/` package running as a background asyncio task inside the
existing FastAPI process (started/stopped from `app/main.py`'s `lifespan`). A long-polling
loop pulls `Update`s from the MAX Bot API and routes them to command handlers; an
`AsyncIOScheduler` drives periodic jobs (FBO slot/coefficient polling, FBS deadline checks).
Bot-specific operational state (active watches, sent-alert dedup) lives in a local SQLite
file, kept separate from the read-only ClickHouse replica the rest of the app uses.

**Tech Stack:** Python, `httpx` (already a dependency) for all three external HTTP APIs,
stdlib `sqlite3` for bot state, `APScheduler` (new dependency) for scheduling, `pytest` +
`httpx.MockTransport` for tests (no `respx`/`tenacity` — avoiding new dependencies where a
few lines of stdlib code suffice).

**Spec:** `docs/superpowers/specs/2026-09-13-max-bot-design.md`

## Global Constraints

- MAX Bot API transport is **long polling** (`GET https://platform-api2.max.ru/updates`),
  not webhook — per spec, no public HTTPS endpoint available for this deployment.
- Auth header to MAX is the raw token: `Authorization: <token>` (no `Bearer` prefix).
- The bot only acts on messages from the one configured `MAX_CHAT_ID` — everything else is
  ignored.
- Ozon FBO: **never auto-book a slot with `coefficient > 0`** (paid) — always ask for
  explicit confirmation in chat first. Only `coefficient == 0` (free) slots are booked
  automatically.
- Wildberries FBO: **no auto-booking in v1** — only coefficient alerts. Do not implement a
  WB slot-booking call.
- FBS deadline risk is a **threshold check** (hours remaining vs. a configured cutoff), not
  an exact fine calculation.
- Credentials live in `.env`/`Settings`, never in the SQLite state file or in git.

```python
import os
import re
from dataclasses import dataclass
from typing import Mapping

_OZON_PATTERN = re.compile(r"^OZON_CABINET_(\d+)_(NAME|CLIENT_ID|API_KEY)$")
_WB_PATTERN = re.compile(r"^WB_CABINET_(\d+)_(NAME|TOKEN)$")


@dataclass(frozen=True)
class OzonCabinet:
    cabinet_id: int
    name: str
    client_id: str
    api_key: str


@dataclass(frozen=True)
class WbCabinet:
    cabinet_id: int
    name: str
    token: str


def _group_by_cabinet(env: Mapping[str, str], pattern: re.Pattern) -> dict[int, dict[str, str]]:
    grouped: dict[int, dict[str, str]] = {}
    for key, value in env.items():
        match = pattern.match(key)
        if not match:
            continue
        cabinet_id = int(match.group(1))
        grouped.setdefault(cabinet_id, {})[match.group(2)] = value
    return grouped


def load_ozon_cabinets(env: Mapping[str, str] | None = None) -> list[OzonCabinet]:
    env = env if env is not None else os.environ
    grouped = _group_by_cabinet(env, _OZON_PATTERN)

    cabinets = []
    for cabinet_id, values in sorted(grouped.items()):
        if not {"NAME", "CLIENT_ID", "API_KEY"} <= values.keys():
            raise ValueError(f"OZON_CABINET_{cabinet_id} is missing required fields")
        cabinets.append(
            OzonCabinet(
                cabinet_id=cabinet_id,
                name=values["NAME"],
                client_id=values["CLIENT_ID"],
                api_key=values["API_KEY"],
            )
        )
    return cabinets


def load_wb_cabinets(env: Mapping[str, str] | None = None) -> list[WbCabinet]:
    env = env if env is not None else os.environ
    grouped = _group_by_cabinet(env, _WB_PATTERN)

    cabinets = []
    for cabinet_id, values in sorted(grouped.items()):
        if not {"NAME", "TOKEN"} <= values.keys():
            raise ValueError(f"WB_CABINET_{cabinet_id} is missing required fields")
        cabinets.append(WbCabinet(cabinet_id=cabinet_id, name=values["NAME"], token=values["TOKEN"]))
    return cabinets
```

```python
import sqlite3

SCHEMA = """
CREATE TABLE IF NOT EXISTS watch_requests (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    marketplace TEXT NOT NULL,
    cabinet_id INTEGER NOT NULL,
    kind TEXT NOT NULL,
    reference_json TEXT NOT NULL,
    date_from TEXT NOT NULL,
    date_to TEXT NOT NULL,
    max_coefficient INTEGER NOT NULL DEFAULT 0,
    status TEXT NOT NULL DEFAULT 'watching',
    requested_by_chat_id INTEGER NOT NULL,
    created_at TEXT NOT NULL,
    updated_at TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS alert_log (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    marketplace TEXT NOT NULL,
    cabinet_id INTEGER NOT NULL,
    order_key TEXT NOT NULL,
    alert_type TEXT NOT NULL,
    sent_at TEXT NOT NULL,
    UNIQUE(marketplace, cabinet_id, order_key, alert_type)
);
"""


def connect(path: str) -> sqlite3.Connection:
    conn = sqlite3.connect(path)
    conn.row_factory = sqlite3.Row
    conn.executescript(SCHEMA)
    return conn


def create_watch_request(
    conn: sqlite3.Connection,
    *,
    marketplace: str,
    cabinet_id: int,
    kind: str,
    reference_json: str,
    date_from: str,
    date_to: str,
    max_coefficient: int,
    requested_by_chat_id: int,
    now: str,
) -> int:
    cursor = conn.execute(
        """
        INSERT INTO watch_requests (
            marketplace, cabinet_id, kind, reference_json, date_from, date_to,
            max_coefficient, status, requested_by_chat_id, created_at, updated_at
        ) VALUES (?, ?, ?, ?, ?, ?, ?, 'watching', ?, ?, ?)
        """,
        (
            marketplace, cabinet_id, kind, reference_json, date_from, date_to,
            max_coefficient, requested_by_chat_id, now, now,
        ),
    )
    conn.commit()
    return cursor.lastrowid


def list_watch_requests(conn: sqlite3.Connection, *, status: str) -> list[sqlite3.Row]:
    return conn.execute(
        "SELECT * FROM watch_requests WHERE status = ? ORDER BY id", (status,)
    ).fetchall()


def update_watch_status(conn: sqlite3.Connection, watch_id: int, status: str, now: str) -> None:
    conn.execute(
        "UPDATE watch_requests SET status = ?, updated_at = ? WHERE id = ?",
        (status, now, watch_id),
    )
    conn.commit()


def record_alert(
    conn: sqlite3.Connection,
    *,
    marketplace: str,
    cabinet_id: int,
    order_key: str,
    alert_type: str,
    now: str,
) -> bool:
    try:
        conn.execute(
            """
            INSERT INTO alert_log (marketplace, cabinet_id, order_key, alert_type, sent_at)
            VALUES (?, ?, ?, ?, ?)
            """,
            (marketplace, cabinet_id, order_key, alert_type, now),
        )
        conn.commit()
        return True
    except sqlite3.IntegrityError:
        return False
```

```python
import httpx

BASE_URL = "https://platform-api2.max.ru"


class MaxClient:
    def __init__(self, token: str, client: httpx.Client | None = None):
        self._client = client or httpx.Client(
            base_url=BASE_URL, headers={"Authorization": token}
        )

    def get_updates(self, marker: int | None = None, timeout: int = 30, limit: int = 100) -> dict:
        params = {"timeout": timeout, "limit": limit}
        if marker is not None:
            params["marker"] = marker
        resp = self._client.get("/updates", params=params, timeout=timeout + 10)
        resp.raise_for_status()
        return resp.json()

    def send_message(
        self, chat_id: int, text: str, buttons: list[list[dict]] | None = None
    ) -> dict:
        payload: dict = {"text": text}
        if buttons:
            payload["attachments"] = [
                {"type": "inline_keyboard", "payload": {"buttons": buttons}}
            ]
        resp = self._client.post("/messages", params={"chat_id": chat_id}, json=payload)
        resp.raise_for_status()
        return resp.json()

    def answer_callback(self, callback_id: str, text: str | None = None) -> dict:
        payload: dict = {}
        if text:
            payload["message"] = {"text": text}
        resp = self._client.post("/answers", params={"callback_id": callback_id}, json=payload)
        resp.raise_for_status()
        return resp.json()
```

```python
import httpx

BASE_URL = "https://api-seller.ozon.ru"


class OzonClient:
    def __init__(self, client_id: str, api_key: str, client: httpx.Client | None = None):
        self._client = client or httpx.Client(
            base_url=BASE_URL,
            headers={"Client-Id": client_id, "Api-Key": api_key},
        )

    def list_unfulfilled_postings(self, limit: int = 1000) -> list[dict]:
        postings: list[dict] = []
        offset = 0
        while True:
            resp = self._client.post(
                "/v3/posting/fbs/unfulfilled/list",
                json={
                    "dir": "asc",
                    "filter": {},
                    "limit": limit,
                    "offset": offset,
                    "with": {"analytics_data": False},
                },
            )
            resp.raise_for_status()
            result = resp.json()["result"]
            postings.extend(result["postings"])
            if not result.get("has_next"):
                break
            offset += limit
        return postings
```

```python
import datetime
from dataclasses import dataclass


@dataclass(frozen=True)
class FbsSnapshot:
    marketplace: str
    cabinet_id: int
    pending_count: int
    earliest_deadline: datetime.datetime | None


def ozon_order_deadlines(postings: list[dict]) -> dict[str, datetime.datetime]:
    deadlines = {}
    for posting in postings:
        shipment_date = posting.get("shipment_date")
        posting_number = posting.get("posting_number")
        if shipment_date and posting_number:
            deadlines[posting_number] = datetime.datetime.fromisoformat(shipment_date)
    return deadlines


def wb_order_deadlines(orders: list[dict], assembly_hours: float) -> dict[str, datetime.datetime]:
    # WB's orders/new response has no documented deadline field (see spec's "Открытые
    # вопросы") — approximated client-side as createdAt + assembly_hours.
    deadlines = {}
    for order in orders:
        created_at = order.get("createdAt")
        order_uid = order.get("orderUid")
        if created_at and order_uid:
            deadlines[order_uid] = datetime.datetime.fromisoformat(created_at) + datetime.timedelta(
                hours=assembly_hours
            )
    return deadlines


def summarize(
    marketplace: str, cabinet_id: int, deadlines: dict[str, datetime.datetime], pending_count: int
) -> FbsSnapshot:
    return FbsSnapshot(
        marketplace=marketplace,
        cabinet_id=cabinet_id,
        pending_count=pending_count,
        earliest_deadline=min(deadlines.values()) if deadlines else None,
    )


def format_status_message(snapshots: list[FbsSnapshot], now: datetime.datetime) -> str:
    lines = ["Статус ФБС-заказов:"]
    for snap in snapshots:
        label = f"{snap.marketplace} (кабинет {snap.cabinet_id})"
        if snap.pending_count == 0:
            lines.append(f"{label}: нет заказов к отгрузке")
            continue
        if snap.earliest_deadline is None:
            lines.append(f"{label}: к отгрузке {snap.pending_count}")
            continue
        hours_left = (snap.earliest_deadline - now).total_seconds() / 3600
        lines.append(
            f"{label}: к отгрузке {snap.pending_count}, ближайший дедлайн через {hours_left:.1f} ч"
        )
    return "\n".join(lines)


def orders_at_risk(
    deadlines: dict[str, datetime.datetime], now: datetime.datetime, threshold_hours: float
) -> dict[str, datetime.datetime]:
    cutoff = now + datetime.timedelta(hours=threshold_hours)
    return {key: deadline for key, deadline in deadlines.items() if deadline <= cutoff}
```

```python
import datetime
import sqlite3

from app.bot import state
from app.bot.fbs_status import orders_at_risk


def check_fbs_deadlines(
    conn: sqlite3.Connection,
    *,
    marketplace: str,
    cabinet_id: int,
    deadlines: dict[str, datetime.datetime],
    now: datetime.datetime,
    threshold_hours: float,
) -> list[str]:
    at_risk = orders_at_risk(deadlines, now, threshold_hours)
    messages = []
    for order_key, deadline in at_risk.items():
        is_new = state.record_alert(
            conn,
            marketplace=marketplace,
            cabinet_id=cabinet_id,
            order_key=order_key,
            alert_type="deadline_risk",
            now=now.isoformat(),
        )
        if not is_new:
            continue
        hours_left = (deadline - now).total_seconds() / 3600
        messages.append(
            f"⚠ {marketplace} (кабинет {cabinet_id}): заказ {order_key} — "
            f"до дедлайна {hours_left:.1f} ч, риск не успеть в доставку"
        )
    return messages
```

```python
import datetime

from app.bot.fbs_status import (
    format_status_message,
    ozon_order_deadlines,
    summarize,
    wb_order_deadlines,
)


def parse_update(update: dict) -> dict | None:
    update_type = update.get("update_type")

    if update_type == "message_created":
        message = update["message"]
        text = ((message.get("body") or {}).get("text") or "").strip()
        return {"kind": "command", "chat_id": message["recipient"]["chat_id"], "text": text}

    if update_type == "message_callback":
        callback = update["callback"]
        message = update.get("message") or {}
        return {
            "kind": "callback",
            "chat_id": (message.get("recipient") or {}).get("chat_id"),
            "callback_id": callback["callback_id"],
            "payload": callback.get("payload", ""),
        }

    return None


def handle_status_command(
    max_client,
    *,
    chat_id: int,
    cabinets_data: list[tuple[str, int, list[dict]]],
    now: datetime.datetime,
    wb_assembly_hours: float = 48.0,
) -> None:
    snapshots = []
    for marketplace, cabinet_id, raw_items in cabinets_data:
        if marketplace == "ozon":
            deadlines = ozon_order_deadlines(raw_items)
        else:
            deadlines = wb_order_deadlines(raw_items, assembly_hours=wb_assembly_hours)
        snapshots.append(summarize(marketplace, cabinet_id, deadlines, pending_count=len(raw_items)))

    max_client.send_message(chat_id, format_status_message(snapshots, now))
```

```python
import json
import sqlite3

from app.bot import state


def _pick_free_slot(timeslots: list[dict], max_coefficient: int) -> dict | None:
    candidates = [t for t in timeslots if t["coefficient"] <= max_coefficient]
    return candidates[0] if candidates else None


def _cheapest_paid_slot(timeslots: list[dict]) -> dict | None:
    paid = [t for t in timeslots if t["coefficient"] > 0]
    return min(paid, key=lambda t: t["coefficient"]) if paid else None


def start_draft_booking(
    ozon_client,
    conn: sqlite3.Connection,
    *,
    cabinet_id: int,
    items: list[dict],
    warehouse_ids: list[int],
    date_from: str,
    date_to: str,
    chat_id: int,
    now: str,
) -> dict:
    operation_id = ozon_client.create_draft(items, warehouse_ids)
    draft_id = ozon_client.get_draft_id(operation_id)
    timeslots = ozon_client.get_draft_timeslots(draft_id, date_from, date_to) if draft_id else []

    free_slot = _pick_free_slot(timeslots, max_coefficient=0)
    if free_slot is not None:
        ozon_client.book_draft_timeslot(draft_id, free_slot["timeslot"])
        return {
            "outcome": "booked",
            "message": f"Забронирован бесплатный слот {free_slot['date']} (склад {free_slot['warehouse_id']})",
        }

    paid_slot = _cheapest_paid_slot(timeslots)
    if paid_slot is not None:
        return {
            "outcome": "awaiting_confirmation",
            "message": f"Найден платный слот {paid_slot['date']} (коэффициент {paid_slot['coefficient']}) — нужно подтверждение",
            "draft_id": draft_id,
            "timeslot": paid_slot["timeslot"],
        }

    watch_id = state.create_watch_request(
        conn,
        marketplace="ozon",
        cabinet_id=cabinet_id,
        kind="ozon_draft_create",
        reference_json=json.dumps({"draft_id": draft_id}),
        date_from=date_from,
        date_to=date_to,
        max_coefficient=0,
        requested_by_chat_id=chat_id,
        now=now,
    )
    return {
        "outcome": "watching",
        "message": "Свободных слотов пока нет, продолжаю наблюдение",
        "watch_id": watch_id,
    }


def poll_draft_watch(ozon_client, conn: sqlite3.Connection, watch_row, now: str) -> str | None:
    reference = json.loads(watch_row["reference_json"])
    draft_id = reference["draft_id"]

    timeslots = ozon_client.get_draft_timeslots(
        draft_id, watch_row["date_from"], watch_row["date_to"]
    )
    free_slot = _pick_free_slot(timeslots, max_coefficient=watch_row["max_coefficient"])
    if free_slot is None:
        return None

    ozon_client.book_draft_timeslot(draft_id, free_slot["timeslot"])
    state.update_watch_status(conn, watch_row["id"], "booked", now)
    return f"Забронирован бесплатный слот {free_slot['date']} (склад {free_slot['warehouse_id']})"
```

```python
import datetime
import json
import sqlite3

from app.bot import state


def start_coefficient_watch(
    conn: sqlite3.Connection,
    *,
    cabinet_id: int,
    warehouse_id: int,
    date_from: str,
    date_to: str,
    max_coefficient: int,
    chat_id: int,
    now: str,
) -> int:
    return state.create_watch_request(
        conn,
        marketplace="wb",
        cabinet_id=cabinet_id,
        kind="wb_coefficient",
        reference_json=json.dumps({"warehouse_id": warehouse_id}),
        date_from=date_from,
        date_to=date_to,
        max_coefficient=max_coefficient,
        requested_by_chat_id=chat_id,
        now=now,
    )


def poll_coefficient_watch(wb_client, conn: sqlite3.Connection, watch_row, now: str) -> str | None:
    reference = json.loads(watch_row["reference_json"])
    warehouse_id = reference["warehouse_id"]
    date_from = watch_row["date_from"]
    date_to = watch_row["date_to"]
    max_coefficient = watch_row["max_coefficient"]

    coefficients = wb_client.get_acceptance_coefficients([warehouse_id])
    for entry in coefficients:
        if entry["warehouseID"] != warehouse_id:
            continue
        if entry["coefficient"] > max_coefficient:
            continue
        entry_date = entry["date"][:10]
        if not (date_from <= entry_date <= date_to):
            continue

        is_new = state.record_alert(
            conn,
            marketplace="wb",
            cabinet_id=watch_row["cabinet_id"],
            order_key=f"{warehouse_id}:{entry_date}",
            alert_type="coefficient_available",
            now=now,
        )
        if not is_new:
            return None
        return (
            f"Появился слот на складе {warehouse_id} ({entry_date}), "
            f"коэффициент {entry['coefficient']} — забронируйте в личном кабинете"
        )
    return None
```

Note on the cooldown test: `record_alert`'s uniqueness key is
`(marketplace, cabinet_id, order_key, alert_type)` with no time component, so once an alert
fires for a given `(warehouse, date)` it never fires again — not just within a cooldown
window. That matches "не чаще настроенного периода" only in the degenerate case where the

```python
import sqlite3

from apscheduler.schedulers.asyncio import AsyncIOScheduler

from app.bot import state
from app.bot.booking_ozon import poll_draft_watch
from app.bot.watch_wb import poll_coefficient_watch


def poll_all_watches(
    ozon_clients: dict[int, object],
    wb_clients: dict[int, object],
    conn: sqlite3.Connection,
    now: str,
    cooldown_minutes: float,
) -> list[str]:
    messages = []
    for row in state.list_watch_requests(conn, status="watching"):
        if row["marketplace"] == "ozon":
            client = ozon_clients.get(row["cabinet_id"])
            if client is None:
                continue
            message = poll_draft_watch(client, conn, row, now)
        else:
            client = wb_clients.get(row["cabinet_id"])
            if client is None:
                continue
            message = poll_coefficient_watch(client, conn, row, now, cooldown_minutes)
        if message is not None:
            messages.append(message)
    return messages


def sweep_expired_watches(conn: sqlite3.Connection, now: str) -> list[int]:
    expired_ids = []
    for row in state.list_watch_requests(conn, status="watching"):
        if row["date_to"] < now[:10]:
            state.update_watch_status(conn, row["id"], "expired", now)
            expired_ids.append(row["id"])
    return expired_ids


def build_scheduler(
    *,
    ozon_clients: dict[int, object],
    wb_clients: dict[int, object],
    conn: sqlite3.Connection,
    max_client,
    chat_id: int,
    watch_poll_seconds: int,
    fbs_poll_minutes: int,
    cooldown_minutes: float,
    now_fn,
) -> AsyncIOScheduler:
    scheduler = AsyncIOScheduler()

    def _poll_watches_job():
        messages = poll_all_watches(ozon_clients, wb_clients, conn, now_fn().isoformat(), cooldown_minutes)
        messages.extend(f"Наблюдение истекло: watch {wid}" for wid in sweep_expired_watches(conn, now_fn().isoformat()))
        for message in messages:
            max_client.send_message(chat_id, message)

    scheduler.add_job(_poll_watches_job, "interval", seconds=watch_poll_seconds, id="poll_watches")
    return scheduler
```

```python
import datetime
from dataclasses import dataclass
from typing import Callable

from app.bot.handlers import parse_update
from app.bot.max_client import MaxClient


def poll_once(
    max_client: MaxClient,
    *,
    marker: int | None,
    on_command: Callable[[int, str], None],
    on_callback: Callable[[int | None, str, str], None],
) -> int | None:
    response = max_client.get_updates(marker=marker)
    for update in response.get("updates", []):
        parsed = parse_update(update)
        if parsed is None:
            continue
        if parsed["kind"] == "command":
            on_command(parsed["chat_id"], parsed["text"])
        elif parsed["kind"] == "callback":
            on_callback(parsed["chat_id"], parsed["callback_id"], parsed["payload"])
    return response.get("marker", marker)


@dataclass
class BotContext:
    max_client: MaxClient
    conn: object
    ozon_clients: dict
    wb_clients: dict
    chat_id: int
    scheduler: object


def build_context(settings) -> BotContext:
    from app.bot import state
    from app.bot.cabinets import load_ozon_cabinets, load_wb_cabinets
    from app.bot.ozon_client import OzonClient
    from app.bot.wb_client import WbClient
    from app.bot.scheduler import build_scheduler

    conn = state.connect("bot_state.db")
    max_client = MaxClient(token=settings.max_bot_token)
    ozon_clients = {
        c.cabinet_id: OzonClient(client_id=c.client_id, api_key=c.api_key)
        for c in load_ozon_cabinets()
    }
    wb_clients = {c.cabinet_id: WbClient(token=c.token) for c in load_wb_cabinets()}
    scheduler = build_scheduler(
        ozon_clients=ozon_clients,
        wb_clients=wb_clients,
        conn=conn,
        max_client=max_client,
        chat_id=settings.max_chat_id,
        watch_poll_seconds=settings.ozon_watch_poll_interval_seconds,
        fbs_poll_minutes=settings.fbs_poll_interval_minutes,
        cooldown_minutes=settings.wb_coefficient_alert_cooldown_minutes,
        now_fn=datetime.datetime.now,
    )
    return BotContext(
        max_client=max_client, conn=conn, ozon_clients=ozon_clients, wb_clients=wb_clients,
        chat_id=settings.max_chat_id, scheduler=scheduler,
    )


async def run_forever(context: BotContext) -> None:
    from app.bot.handlers import handle_cancel_command, handle_status_command

    context.scheduler.start()
    marker = None
    try:
        while True:
            def on_command(chat_id, text):
                if chat_id != context.chat_id:
                    return
                if text == "/status":
                    cabinets_data = [
                        ("ozon", cid, client.list_unfulfilled_postings())
                        for cid, client in context.ozon_clients.items()
                    ] + [
                        ("wb", cid, client.list_new_orders())
                        for cid, client in context.wb_clients.items()
                    ]
                    handle_status_command(
                        context.max_client, chat_id=chat_id, cabinets_data=cabinets_data,
                        now=datetime.datetime.now(datetime.timezone.utc),
                    )
                elif text.startswith("/cancel"):
                    reply = handle_cancel_command(
                        context.conn, chat_id=chat_id, args=text.split()[1:],
                        now=datetime.datetime.now(datetime.timezone.utc).isoformat(),
                    )
                    context.max_client.send_message(chat_id, reply)

            def on_callback(chat_id, callback_id, payload):
                context.max_client.answer_callback(callback_id, text="Принято")

            marker = poll_once(
                context.max_client, marker=marker, on_command=on_command, on_callback=on_callback
            )
    finally:
        context.scheduler.shutdown(wait=False)
```
